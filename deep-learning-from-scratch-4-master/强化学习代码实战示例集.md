## 示例 1：多臂老虎机（探索与利用）

> **场景**：5 台老虎机，每台中奖概率不同。算法要在 1000 次尝试中学会"挑概率最高的"。

```python
"""
示例 1：多臂老虎机（Multi-Armed Bandit）
算法：ε-greedy（贪心 + 随机探索）
对应原书：ch01/bandit.py
"""

import numpy as np
import matplotlib.pyplot as plt


class Bandit:
    """5 台老虎机，每台中奖概率不同"""

    def __init__(self, n_arms=5):
        # 真实的中奖概率（算法不知道！）
        self.probs = np.random.rand(n_arms)
        # 每台老虎机的"估计中奖概率"（算法维护）
        self.q_values = np.zeros(n_arms)
        # 每台被尝试的次数
        self.counts = np.zeros(n_arms)

    def pull(self, arm):
        """拉一下某台老虎机，返回 1（中奖）或 0（没中）"""
        return 1 if np.random.rand() < self.probs[arm] else 0

    def update(self, arm, reward):
        """增量更新这台老虎机的估计值（核心公式）"""
        self.counts[arm] += 1
        # 增量平均公式：Q_new = Q_old + (1/n) * (reward - Q_old)
        self.q_values[arm] += (reward - self.q_values[arm]) / self.counts[arm]


def epsilon_greedy(bandit, epsilon=0.1, n_steps=1000):
    """ε-greedy 策略：以概率 ε 随机探索，否则选当前最优"""
    rewards = []
    for _ in range(n_steps):
        # 探索 vs 利用
        if np.random.rand() < epsilon:
            arm = np.random.randint(len(bandit.probs))   # 随机选
        else:
            arm = np.argmax(bandit.q_values)             # 选当前最优
        # 执行 + 学习
        reward = bandit.pull(arm)
        bandit.update(arm, reward)
        rewards.append(reward)
    return rewards


def ucb(bandit, c=2.0, n_steps=1000):
    """UCB（Upper Confidence Bound）：乐观面对不确定性"""
    rewards = []
    for t in range(1, n_steps + 1):
        # 优先尝试还没拉过的老虎机
        untried = np.where(bandit.counts == 0)[0]
        if len(untried) > 0:
            arm = untried[0]
        else:
            # UCB 公式：Q + c * sqrt(ln(t) / n)
            ucb_values = bandit.q_values + c * np.sqrt(np.log(t) / bandit.counts)
            arm = np.argmax(ucb_values)
        reward = bandit.pull(arm)
        bandit.update(arm, reward)
        rewards.append(reward)
    return rewards


# ===== 主程序 =====
if __name__ == "__main__":
    # 跑 100 次实验取平均，看哪种策略更稳
    eps_rewards, ucb_rewards = [], []
    for _ in range(100):
        b1, b2 = Bandit(), Bandit()
        eps_rewards.append(np.cumsum(epsilon_greedy(b1, epsilon=0.1)))
        ucb_rewards.append(np.cumsum(ucb(b2, c=2.0)))

    plt.plot(np.mean(eps_rewards, axis=0), label="ε-greedy (ε=0.1)")
    plt.plot(np.mean(ucb_rewards, axis=0), label="UCB (c=2.0)")
    plt.xlabel("Steps")
    plt.ylabel("Cumulative Reward")
    plt.legend()
    plt.title("Bandit: ε-greedy vs UCB")
    plt.show()
```

**核心心法**：
- **增量更新**：`Q_new = Q_old + (reward - Q_old) / n`
- **探索 vs 利用**：ε-greedy 用概率，UCB 用"不确定性"

## 示例 2：FrozenLake 表格 Q-Learning

> **场景**：4×4 冰面，踩到洞就掉下去，走到目标就赢。用 Q 表（状态-动作）学最优策略。

```python
"""
示例 2：FrozenLake 表格 Q-Learning
算法：表格型 Q-Learning（状态、动作都有限）
对应原书：ch06/q_learning.py
"""

import numpy as np
import gymnasium as gym
from collections import defaultdict


class QLearningAgent:
    """Q-Learning 智能体：维护一张 Q[state][action] 表"""

    def __init__(self, n_actions, alpha=0.1, gamma=0.99, epsilon=0.1):
        self.n_actions = n_actions
        self.alpha = alpha   # 学习率
        self.gamma = gamma   # 折扣因子
        self.epsilon = epsilon
        # Q 表：state -> {action: q_value}
        self.Q = defaultdict(lambda: np.zeros(n_actions))

    def get_action(self, state):
        """ε-greedy 选动作"""
        if np.random.rand() < self.epsilon:
            return np.random.randint(self.n_actions)
        return np.argmax(self.Q[state])

    def update(self, state, action, reward, next_state, done):
        """Q-Learning 核心更新公式"""
        # 目标 = 即时奖励 + γ * max_a' Q(s', a')   （如果没结束）
        if done:
            target = reward
        else:
            target = reward + self.gamma * np.max(self.Q[next_state])
        # 增量更新：Q(s,a) ← Q(s,a) + α * (target - Q(s,a))
        self.Q[state][action] += self.alpha * (target - self.Q[state][action])


def train(n_episodes=5000):
    env = gym.make("FrozenLake-v1", is_slippery=False)  # 关闭滑冰，便于学习
    agent = QLearningAgent(n_actions=env.action_space.n)
    rewards_history = []

    for ep in range(n_episodes):
        state, _ = env.reset()
        total_reward = 0
        done = False

        while not done:
            action = agent.get_action(state)
            next_state, reward, terminated, truncated, _ = env.step(action)
            done = terminated or truncated
            agent.update(state, action, reward, next_state, done)
            state = next_state
            total_reward += reward

        rewards_history.append(total_reward)

    # 打印最近 100 回合的胜率
    win_rate = np.mean(rewards_history[-100:])
    print(f"最近 100 回合胜率：{win_rate:.2%}")
    return agent


# ===== 测试学到的策略 =====
def test(agent, n_episodes=100):
    env = gym.make("FrozenLake-v1", is_slippery=False)
    wins = 0
    for _ in range(n_episodes):
        state, _ = env.reset()
        done = False
        while not done:
            action = np.argmax(agent.Q[state])   # 学完后不再探索
            state, reward, terminated, truncated, _ = env.step(action)
            done = terminated or truncated
            if reward == 1:
                wins += 1
    print(f"测试胜率：{wins / n_episodes:.2%}")


if __name__ == "__main__":
    agent = train()
    test(agent)
```

**核心心法**：
- **Bellman 方程**：`Q(s,a) ← Q(s,a) + α * [r + γ * max_a' Q(s',a') - Q(s,a)]`
- **off-policy**：学习时用 `max`（假设未来用最优策略），但行动时仍可探索。

## 示例 3：CartPole DQN（经典案例）

> **场景**：让小车左右移动，保持杆子不倒。这是 DQN 的"Hello World"。

```python
"""
示例 3：CartPole DQN
算法：DQN（Deep Q-Network）
对应原书：ch08/dqn.py
关键技巧：经验回放 + 目标网络
"""

import random
from collections import deque, namedtuple
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
import gymnasium as gym


# ===== 经验回放（Replay Buffer）=====
Transition = namedtuple("Transition", ["state", "action", "reward", "next_state", "done"])

class ReplayBuffer:
    """存 10000 条经验，训练时随机采样（打破数据相关性）"""

    def __init__(self, capacity=10000):
        self.buffer = deque(maxlen=capacity)

    def push(self, *args):
        self.buffer.append(Transition(*args))

    def sample(self, batch_size):
        batch = random.sample(self.buffer, batch_size)
        # 把列表拆成 5 个张量
        s = torch.FloatTensor(np.array([t.state for t in batch]))
        a = torch.LongTensor([t.action for t in batch]).unsqueeze(1)
        r = torch.FloatTensor([t.reward for t in batch])
        s_next = torch.FloatTensor(np.array([t.next_state for t in batch]))
        d = torch.FloatTensor([t.done for t in batch])
        return s, a, r, s_next, d

    def __len__(self):
        return len(self.buffer)


# ===== Q 网络 =====
class QNetwork(nn.Module):
    """输入状态，输出每个动作的 Q 值"""

    def __init__(self, state_dim, action_dim, hidden=64):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(state_dim, hidden), nn.ReLU(),
            nn.Linear(hidden, hidden), nn.ReLU(),
            nn.Linear(hidden, action_dim)
        )

    def forward(self, x):
        return self.net(x)


# ===== DQN Agent =====
class DQNAgent:
    def __init__(self, state_dim, action_dim, lr=1e-3, gamma=0.99):
        self.gamma = gamma
        self.q_net = QNetwork(state_dim, action_dim)         # 在线网络
        self.target_net = QNetwork(state_dim, action_dim)     # 目标网络（稳定训练）
        self.target_net.load_state_dict(self.q_net.state_dict())
        self.optimizer = optim.Adam(self.q_net.parameters(), lr=lr)
        self.buffer = ReplayBuffer()

    def get_action(self, state, epsilon=0.1):
        """ε-greedy"""
        if random.random() < epsilon:
            return random.randrange(action_dim)
        with torch.no_grad():
            return self.q_net(torch.FloatTensor(state)).argmax().item()

    def train_step(self, batch_size=64):
        if len(self.buffer) < batch_size:
            return None
        s, a, r, s_next, d = self.buffer.sample(batch_size)

        # 当前 Q 值：Q(s, a)
        q_pred = self.q_net(s).gather(1, a).squeeze()
        # 目标 Q 值：用 target_net 算 r + γ * max_a' Q'(s', a')
        with torch.no_grad():
            q_target = r + self.gamma * self.target_net(s_next).max(1)[0] * (1 - d)

        # 损失 = (Q_pred - Q_target)^2
        loss = nn.MSELoss()(q_pred, q_target)
        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()
        return loss.item()

    def update_target(self):
        """每隔几步把在线网络的参数复制到目标网络"""
        self.target_net.load_state_dict(self.q_net.state_dict())


# ===== 训练 =====
def train(n_episodes=500):
    env = gym.make("CartPole-v1")
    agent = DQNAgent(state_dim=4, action_dim=2)
    rewards = []

    for ep in range(n_episodes):
        state, _ = env.reset()
        total_reward = 0
        done = False
        # 让 ε 随训练慢慢衰减（前期多探索，后期多用学到的）
        epsilon = max(0.01, 0.1 * (0.995 ** ep))

        while not done:
            action = agent.get_action(state, epsilon)
            next_state, reward, terminated, truncated, _ = env.step(action)
            done = terminated or truncated
            agent.buffer.push(state, action, reward, next_state, float(done))
            agent.train_step()
            state = next_state
            total_reward += reward

        # 每 10 个回合同步一次目标网络
        if ep % 10 == 0:
            agent.update_target()
        rewards.append(total_reward)

        if ep % 50 == 0:
            avg = np.mean(rewards[-50:])
            print(f"Episode {ep}, Avg Reward: {avg:.1f}, ε: {epsilon:.3f}")

    return agent, rewards


if __name__ == "__main__":
    agent, rewards = train()
    # 通常 200~300 回合就能稳定在 400+
```

**DQN 三大核心**：
1. **经验回放**（Replay Buffer）：打破时序相关性，让数据独立同分布。
2. **目标网络**（Target Network）：稳定 Q 值目标，防止"自己追自己"。
3. **ε-greedy 衰减**：前期多探索，后期多利用。

## 示例 4：DQN 股票交易优化（**预测优化重点**）

> **场景**：用历史股价训练一个 DQN，让它学会"什么时候买、什么时候卖"以最大化收益。
> **核心思想**：把"对未来价格的预测"转化为"对最优交易动作的学习"。

```python
"""
示例 4：DQN 股票交易优化
算法：DQN
为什么用强化学习？
    - 监督学习只能预测"价格会涨/跌"，但无法告诉你"现在该不该操作"
    - 强化学习直接优化"最终收益"，综合考虑持仓、手续费、风险
"""

import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
from collections import deque, namedtuple
import matplotlib.pyplot as plt


# ===== 1. 模拟股票环境 =====
class StockTradingEnv:
    """
    状态 = 最近 N 天的价格 + 持仓 + 现金
    动作 = 0(持有) / 1(买入) / 2(卖出)
    奖励 = 当前总资产 - 上一步总资产
    """

    def __init__(self, prices, lookback=10, initial_cash=10000):
        self.prices = prices                # 模拟的价格序列
        self.lookback = lookback            # 看多少天的历史
        self.initial_cash = initial_cash
        self.reset()

    def reset(self):
        self.t = self.lookback              # 从第 lookback 天开始交易
        self.cash = self.initial_cash
        self.shares = 0
        self.history = []                   # 记录每天的总资产
        return self._get_state()

    def _get_state(self):
        """状态 = 最近 lookback 天的价格 + [现金比例, 持仓比例]"""
        window = self.prices[self.t - self.lookback : self.t]
        # 归一化：让网络更好学
        norm_window = window / window[0] - 1
        # 当前资产结构
        total_asset = self.cash + self.shares * self.prices[self.t]
        cash_ratio = self.cash / total_asset
        shares_ratio = (self.shares * self.prices[self.t]) / total_asset
        return np.concatenate([norm_window, [cash_ratio, shares_ratio]])

    def step(self, action):
        price = self.prices[self.t]
        prev_asset = self.cash + self.shares * price

        # 0=持有, 1=买入, 2=卖出
        if action == 1 and self.cash > price:           # 买入 1 股（简化版）
            self.shares += 1
            self.cash -= price
        elif action == 2 and self.shares > 0:           # 卖出 1 股
            self.shares -= 1
            self.cash += price

        self.t += 1
        done = self.t >= len(self.prices) - 1
        next_state = self._get_state()
        new_asset = self.cash + self.shares * self.prices[self.t]
        # 奖励 = 资产变化（鼓励赚更多钱）
        reward = (new_asset - prev_asset) / prev_asset
        self.history.append(new_asset)
        return next_state, reward, done


# ===== 2. 生成模拟股价（几何布朗运动）=====
def generate_stock_prices(n_days=1000, start=100, mu=0.0005, sigma=0.02, seed=42):
    """模拟一个有点趋势、有点波动的股票"""
    np.random.seed(seed)
    returns = np.random.normal(mu, sigma, n_days)
    prices = start * np.exp(np.cumsum(returns))
    return prices


# ===== 3. DQN 网络（与示例 3 类似，此处简洁）=====
Transition = namedtuple("Transition", ["s", "a", "r", "s_next", "d"])

class StockDQN(nn.Module):
    def __init__(self, state_dim, action_dim=3):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(state_dim, 64), nn.ReLU(),
            nn.Linear(64, 64), nn.ReLU(),
            nn.Linear(64, action_dim)
        )
    def forward(self, x):
        return self.net(x)


class StockAgent:
    def __init__(self, state_dim):
        self.q = StockDQN(state_dim)
        self.target = StockDQN(state_dim)
        self.target.load_state_dict(self.q.state_dict())
        self.opt = optim.Adam(self.q.parameters(), lr=1e-3)
        self.buf = deque(maxlen=5000)
        self.gamma = 0.95

    def act(self, state, eps=0.1):
        if np.random.rand() < eps:
            return np.random.randint(3)
        with torch.no_grad():
            return self.q(torch.FloatTensor(state)).argmax().item()

    def train(self, batch=32):
        if len(self.buf) < batch:
            return
        batch_data = random.sample(self.buf, batch)
        s = torch.FloatTensor([t.s for t in batch_data])
        a = torch.LongTensor([t.a for t in batch_data]).unsqueeze(1)
        r = torch.FloatTensor([t.r for t in batch_data])
        s_next = torch.FloatTensor([t.s_next for t in batch_data])
        d = torch.FloatTensor([t.d for t in batch_data])

        q_pred = self.q(s).gather(1, a).squeeze()
        with torch.no_grad():
            q_next = self.target(s_next).max(1)[0]
            q_target = r + self.gamma * q_next * (1 - d)
        loss = nn.MSELoss()(q_pred, q_target)
        self.opt.zero_grad()
        loss.backward()
        self.opt.step()


import random


# ===== 4. 训练 + 回测 =====
def train_and_backtest(n_episodes=100):
    prices = generate_stock_prices()
    env = StockTradingEnv(prices)
    state_dim = env.lookback + 2          # lookback 天价格 + 2 个资产比例
    agent = StockAgent(state_dim)

    # ---------- 训练 ----------
    for ep in range(n_episodes):
        state = env.reset()
        done = False
        while not done:
            action = agent.act(state, eps=0.1)
            next_state, reward, done = env.step(action)
            agent.buf.append(Transition(state, action, reward, next_state, float(done)))
            agent.train()
            state = next_state
        # 每 5 个回合同步目标网络
        if ep % 5 == 0:
            agent.target.load_state_dict(agent.q.state_dict())

    # ---------- 回测（无探索）----------
    env = StockTradingEnv(prices)
    state = env.reset()
    done = False
    while not done:
        action = agent.act(state, eps=0.0)    # 用学到的策略
        state, _, done = env.step(action)

    # 对比：买入并持有策略
    buy_and_hold = prices[-1] / prices[env.lookback] * env.initial_cash

    print(f"DQN 最终资产：   ${env.history[-1]:.2f}")
    print(f"买入持有策略：   ${buy_and_hold:.2f}")
    print(f"DQN 收益率：{(env.history[-1] / env.initial_cash - 1):.2%}")
    print(f"持有收益率：{(buy_and_hold / env.initial_cash - 1):.2%}")

    # 画资产曲线
    plt.plot(env.history, label="DQN Strategy")
    plt.axhline(buy_and_hold, color="r", linestyle="--", label="Buy & Hold")
    plt.xlabel("Days")
    plt.ylabel("Total Asset ($)")
    plt.legend()
    plt.title("Stock Trading: DQN vs Buy & Hold")
    plt.show()


if __name__ == "__main__":
    train_and_backtest()
```

### 关键设计点（为什么这样设计？）

| 设计 | 原因 |
| --- | --- |
| 状态包含**历史价格 + 持仓** | 让智能体能同时"看趋势"和"看自己仓位" |
| 奖励 = **资产变化率** | 鼓励相对收益，而非绝对金额（不同起点可比） |
| 动作 = **买/卖/持有** | 比"预测价格"更接近最终目标 |
| 目标网络 + 回放 | 金融数据高噪声，必须稳定训练 |

## 示例 5：DQN 动态定价优化

> **场景**：电商每天要给商品定价，价格高 → 销量低，价格低 → 销量高但利润薄。DQN 学"怎么定价最赚"。

```python
"""
示例 5：DQN 动态定价优化
算法：DQN
应用：电商动态定价、机票定价、共享单车定价
"""

import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
from collections import deque
import random
import matplotlib.pyplot as plt


class PricingEnv:
    """
    状态 = [当前价格, 过去7天销量, 过去7天利润, 库存]
    动作 = 价格档位（0=降价20%, 1=不变, 2=涨价20%）
    奖励 = 当日利润
    """

    def __init__(self, n_days=200):
        self.n_days = n_days
        self.reset()

    def reset(self):
        self.t = 0
        self.price = 100          # 初始定价
        self.inventory = 1000     # 库存
        self.demand_history = deque(maxlen=7)
        self.profit_history = deque(maxlen=7)
        return self._get_state()

    def _demand(self, price):
        """价格越高，需求越少（线性关系，简化）"""
        base_demand = 100
        elasticity = -0.5
        d = base_demand * (1 + elasticity * (price - 100) / 100)
        # 加噪声
        d += np.random.normal(0, 5)
        return max(0, int(d))

    def _get_state(self):
        avg_demand = np.mean(self.demand_history) if self.demand_history else 50
        avg_profit = np.mean(self.profit_history) if self.profit_history else 0
        return np.array([
            self.price / 100,           # 归一化价格
            avg_demand / 100,           # 归一化需求
            avg_profit / 1000,          # 归一化利润
            self.inventory / 1000       # 归一化库存
        ], dtype=np.float32)

    def step(self, action):
        # 动作 → 调价
        price_delta = [-20, 0, 20][action]
        self.price = max(50, min(150, self.price + price_delta))

        # 模拟市场需求
        demand = self._demand(self.price)
        sold = min(demand, self.inventory)
        self.inventory -= sold

        # 利润 = 销售额 - 成本（成本假设 60/件）
        cost = 60
        profit = sold * (self.price - cost)

        self.demand_history.append(demand)
        self.profit_history.append(profit)
        self.t += 1

        done = self.t >= self.n_days or self.inventory <= 0
        return self._get_state(), profit / 100, done   # 奖励归一化


class PricingDQN(nn.Module):
    def __init__(self, state_dim, action_dim=3):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(state_dim, 32), nn.ReLU(),
            nn.Linear(32, 32), nn.ReLU(),
            nn.Linear(32, action_dim)
        )
    def forward(self, x):
        return self.net(x)


class PricingAgent:
    def __init__(self, state_dim):
        self.q = PricingDQN(state_dim)
        self.target = PricingDQN(state_dim)
        self.target.load_state_dict(self.q.state_dict())
        self.opt = optim.Adam(self.q.parameters(), lr=1e-3)
        self.buf = deque(maxlen=3000)

    def act(self, state, eps=0.1):
        if np.random.rand() < eps:
            return np.random.randint(3)
        with torch.no_grad():
            return self.q(torch.FloatTensor(state)).argmax().item()

    def train(self, batch=32):
        if len(self.buf) < batch:
            return
        batch_data = random.sample(self.buf, batch)
        s = torch.FloatTensor([t[0] for t in batch_data])
        a = torch.LongTensor([t[1] for t in batch_data])
        r = torch.FloatTensor([t[2] for t in batch_data])
        s_next = torch.FloatTensor([t[3] for t in batch_data])
        d = torch.FloatTensor([t[4] for t in batch_data])
        q_pred = self.q(s).gather(1, a.unsqueeze(1)).squeeze()
        with torch.no_grad():
            q_target = r + 0.95 * self.target(s_next).max(1)[0] * (1 - d)
        loss = nn.MSELoss()(q_pred, q_target)
        self.opt.zero_grad()
        loss.backward()
        self.opt.step()


def train_pricing(n_episodes=200):
    agent = PricingAgent(state_dim=4)
    profit_curve = []

    for ep in range(n_episodes):
        env = PricingEnv()
        state = env.reset()
        total_profit = 0
        done = False
        while not done:
            action = agent.act(state, eps=0.1)
            next_state, reward, done = env.step(action)
            agent.buf.append((state, action, reward, next_state, float(done)))
            agent.train()
            state = next_state
            total_profit += reward
        if ep % 10 == 0:
            agent.target.load_state_dict(self.q.state_dict())
        profit_curve.append(total_profit)

    # 画学习曲线
    plt.plot(profit_curve)
    plt.xlabel("Episode")
    plt.ylabel("Total Profit")
    plt.title("DQN Pricing Learning Curve")
    plt.show()

    return agent


if __name__ == "__main__":
    agent = train_pricing()
    # 用学到的策略跑一次
    env = PricingEnv()
    s = env.reset()
    done = False
    while not done:
        a = agent.act(s, eps=0.0)
        s, r, done = env.step(a)
    print(f"策略最终日均利润：{np.mean([env.profit_history[i] for i in range(len(env.profit_history))]):.1f}")
```

**DQN 预测优化的通用模式**：

```
环境建模 → 状态设计 → 动作设计 → 奖励设计 → DQN 训练 → 策略回测
   ↑                                              ↓
   └──────────── 调整超参数（γ, ε, lr） ←──────────┘
```

## 示例 6：REINFORCE 策略梯度

> **场景**：直接学一个"策略网络"（输入状态，输出动作概率分布）。常用于连续动作。

```python
"""
示例 6：REINFORCE 策略梯度
算法：REINFORCE（蒙特卡洛策略梯度）
对应原书：ch09/reinforce.py
核心：把"未来总奖励"作为权重，让好动作概率变大、差动作变小
"""

import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
import gymnasium as gym


class PolicyNetwork(nn.Module):
    """策略网络：输入状态，输出动作概率分布"""

    def __init__(self, state_dim, action_dim, hidden=16):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(state_dim, hidden), nn.ReLU(),
            nn.Linear(hidden, action_dim),
            nn.Softmax(dim=-1)
        )

    def forward(self, x):
        return self.net(x)


class REINFORCEAgent:
    def __init__(self, state_dim, action_dim, lr=1e-3, gamma=0.99):
        self.policy = PolicyNetwork(state_dim, action_dim)
        self.opt = optim.Adam(self.policy.parameters(), lr=lr)
        self.gamma = gamma
        # 一个 episode 的轨迹
        self.saved_log_probs = []
        self.rewards = []

    def act(self, state):
        """根据策略网络采样动作"""
        state = torch.FloatTensor(state)
        probs = self.policy(state)
        # 按概率分布采样
        dist = torch.distributions.Categorical(probs)
        action = dist.sample()
        self.saved_log_probs.append(dist.log_prob(action))
        return action.item()

    def train(self):
        """回合结束后一次性更新"""
        R = 0
        policy_loss = []
        returns = []
        # 从后往前算折扣奖励：G_t = r_t + γ * G_{t+1}
        for r in reversed(self.rewards):
            R = r + self.gamma * R
            returns.insert(0, R)
        returns = torch.tensor(returns)
        # 标准化：让训练更稳定
        returns = (returns - returns.mean()) / (returns.std() + 1e-9)

        # 损失 = -log_prob * return   （好的动作 log_prob * 大 return → 损失小）
        for log_prob, R in zip(self.saved_log_probs, returns):
            policy_loss.append(-log_prob * R)
        self.opt.zero_grad()
        loss = torch.cat(policy_loss).sum()
        loss.backward()
        self.opt.step()

        # 清空轨迹
        self.saved_log_probs = []
        self.rewards = []


def train(n_episodes=1000):
    env = gym.make("CartPole-v1")
    agent = REINFORCEAgent(state_dim=4, action_dim=2)
    rewards = []

    for ep in range(n_episodes):
        state, _ = env.reset()
        total_reward = 0
        done = False
        while not done:
            action = agent.act(state)
            state, reward, terminated, truncated, _ = env.step(action)
            done = terminated or truncated
            agent.rewards.append(reward)
            total_reward += reward
        agent.train()
        rewards.append(total_reward)

        if ep % 50 == 0:
            print(f"Episode {ep}, Avg Reward: {np.mean(rewards[-50:]):.1f}")

    return agent


if __name__ == "__main__":
    train()
```

**REINFORCE 核心公式**：

$$
\theta \leftarrow \theta + \alpha \cdot G_t \cdot \nabla_\theta \log \pi_\theta(a_t | s_t)
$$

> "**好动作的概率要变大，差动作的概率要变小**"，力度由 $G_t$ 决定。

## 示例 7：Actor-Critic

> **场景**：REINFORCE 要等回合结束才能学，太慢。Actor-Critic 用 Critic 当"即时评委"，每步都能学。

```python
"""
示例 7：Actor-Critic (A2C)
算法：Actor-Critic
对应原书：ch09/actor_critic.py
Actor = 策略网络（决定动作）
Critic = 价值网络（评估状态有多好）
"""

import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
import gymnasium as gym


class ActorCritic(nn.Module):
    """一个网络，两头输出：策略（Actor）+ 价值（Critic）"""

    def __init__(self, state_dim, action_dim, hidden=64):
        super().__init__()
        self.shared = nn.Sequential(
            nn.Linear(state_dim, hidden), nn.ReLU(),
        )
        self.actor = nn.Linear(hidden, action_dim)         # 输出动作 logits
        self.critic = nn.Linear(hidden, 1)                # 输出状态价值 V(s)

    def forward(self, x):
        h = self.shared(x)
        return self.actor(h), self.critic(h)


class A2CAgent:
    def __init__(self, state_dim, action_dim, lr=3e-4, gamma=0.99):
        self.model = ActorCritic(state_dim, action_dim)
        self.opt = optim.Adam(self.model.parameters(), lr=lr)
        self.gamma = gamma

    def act(self, state):
        state = torch.FloatTensor(state)
        logits, _ = self.model(state)
        probs = nn.functional.softmax(logits, dim=-1)
        dist = torch.distributions.Categorical(probs)
        action = dist.sample()
        return action.item(), dist.log_prob(action)

    def train_step(self, log_prob, value, reward, next_value, done):
        """
        TD 误差 δ = r + γ * V(s') - V(s)
        Actor 损失 = -log_prob * δ   （用 δ 替代 REINFORCE 的 G_t，更稳定）
        Critic 损失 = δ^2
        """
        td_target = reward + self.gamma * next_value * (1 - int(done))
        td_error = td_target - value

        actor_loss = -log_prob * td_error.detach()
        critic_loss = td_error.pow(2)
        loss = actor_loss + critic_loss

        self.opt.zero_grad()
        loss.backward()
        self.opt.step()
        return loss.item()


def train(n_episodes=500):
    env = gym.make("CartPole-v1")
    agent = A2CAgent(state_dim=4, action_dim=2)
    rewards = []

    for ep in range(n_episodes):
        state, _ = env.reset()
        total_reward = 0
        done = False
        while not done:
            action, log_prob = agent.act(state)
            next_state, reward, terminated, truncated, _ = env.step(action)
            done = terminated or truncated

            # 计算 V(s) 和 V(s')
            with torch.no_grad():
                _, value = agent.model(torch.FloatTensor(state))
                _, next_value = agent.model(torch.FloatTensor(next_state))

            agent.train_step(log_prob, value, reward, next_value, done)
            state = next_state
            total_reward += reward
        rewards.append(total_reward)

        if ep % 20 == 0:
            print(f"Episode {ep}, Avg Reward: {np.mean(rewards[-20:]):.1f}")

    return agent


if __name__ == "__main__":
    train()
```

**Actor-Critic 的妙处**：

| 组件 | 角色 | 任务 |
| --- | --- | --- |
| **Actor** | 演员 | 决定动作（策略） |
| **Critic** | 评委 | 评估状态（价值） |
| **TD 误差 δ** | 即时反馈 | 告诉 Actor "这步是好是坏" |

## 综合对比：选哪个算法？

| 场景 | 推荐算法 | 原因 |
| --- | --- | --- |
| **状态/动作都有限（< 1000）** | 表格 Q-Learning | 简单、稳定、可解释 |
| **状态高维（图像、向量）、动作离散** | **DQN** | 工业标准 |
| **动作连续（机器人控制）** | **DDPG / SAC / PPO** | DQN 不擅长 |
| **需要快速、稳定训练** | **Actor-Critic (A2C/PPO)** | 方差小、样本效率高 |
| **预测优化类业务**（定价、库存、交易） | **DQN + 领域工程** | 直接优化最终目标 |
| **想理解 RL 本质** | REINFORCE → Q-Learning → DQN → AC | 由浅入深 |

---

## 超参数速查表

| 参数 | 常见值 | 作用 |
| --- | --- | --- |
| `gamma` (γ) | 0.95 ~ 0.99 | 未来奖励的折扣，越大越"有远见" |
| `epsilon` (ε) | 0.1 → 0.01 衰减 | 探索概率 |
| `lr` (学习率) | 1e-4 ~ 1e-3 | 太大震荡，太小学不动 |
| `batch_size` | 32 ~ 256 | 越大越稳，越大越慢 |
| `buffer_size` | 1万 ~ 100万 | 数据多样性 vs 内存 |
| `target_update_freq` | 10 ~ 1000 步 | 目标网络更新频率 |
